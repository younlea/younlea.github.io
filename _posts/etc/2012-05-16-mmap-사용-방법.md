user단에서 kernel단에서 할당한 address를 곧장 접근할수 있도록 한다.

------------------usr app ----------------------------------

void \*uva;

uva = (char \*) mmap( 0,

MMAP\_SIZE,

PROT\_READ|PROT\_WRITE,

MAP\_SHARED,

fd,

0 );

-------------------driver[kernel]----------------------------

int led\_mmap(struct file \*filp , struct vm\_area\_struct \*vma )

{

vma->vm\_flags |= VM\_RESERVED;

vma->vm\_flags |= VM\_IO;

vma->vm\_page\_prot = pgprot\_noncached(vma->vm\_page\_prot);

if(remap\_pfn\_range(vma,vma->vm\_start, 0xE02001C0 /- PAddress\*- >> PAGE\_SHIFT,  4096 , vma->vm\_page\_prot))

return -EAGAIN;

return 0;

}
