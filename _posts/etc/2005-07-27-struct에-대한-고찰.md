{\
 int test\_int;\
 char test\_char;\
 double test\_double;\
}test\_typedef, \*test\_typedef\_point;

{\

 test\_typedef\_point test\_point[10]; //struct pointer형태로 배열을 만듬

 test[0].test\_int=1;\
