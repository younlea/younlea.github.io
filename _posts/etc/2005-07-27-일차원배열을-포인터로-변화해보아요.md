int main(int argc, char \*argv[])\
{\
 \
 int tmp[10]={0};\
 int i=0;

 for (i=0;i<10 ;i++ )\
 {\
 tmp[i]=i;\
 printf("tmp[i] --- tmp[%d] ----> %d \
" ,i, tmp[i]);\
 printf("\*(tmp+i) --- \*(tmp+%d) ----> %d \
" ,i, \*(tmp+i));\
 printf("i[tmp] --- %d[tmp] ----> %d \
" ,i, i[tmp]);

 /// 번지수 확인 \
 printf("%d, %d\
",i,&tmp[i]); //int 형에 대한 자료 확인\
 printf("%d, %l\
",i,\*(double \*)&i[tmp]);\
 printf("\
");

 }\

}\
