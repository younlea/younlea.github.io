max string 찾기

#include <stdio.h>

#define MAX\_STRING 10000

typedef struct my\_st{

int count;

char string[MAX\_STRING];

}my\_st;

int check\_size(char \*str)

{

int count = 0;

while(str[count] != 0 ) count++;

return count;

}

void my\_strcp(char \*str1, char \*str2, int size)

{

int ss = 0;

while(ss < size)

{

str1[ss]=str2[ss];

ss++;

}

return;

}

void main(int argc, char \*argv[])

{

my\_st max\_string[10];

if(argc != 2)

{

printf("please input test word file name\n");

return;

}

//printf("%s\n",argv[1]);

// read file name

FILE \*fp;

fp =fopen(argv[1],"rw");

if(fp == NULL)

{

printf("ERROR!");

return;

}

//check file size

fseek(fp, 0, SEEK\_END);

long size = ftell(fp);

rewind(fp);

//printf("file size : %d \n" ,size);

for(int i =0; i < 10; i++)

{

max\_string[i].count = 0;

for(int j =0; j< MAX\_STRING; j++)

max\_string[i].string[j] = 0;

}

int s\_size;

int size\_count = 0;

char r\_string[MAX\_STRING]={0,};

int my\_stop = 1;

while(my\_stop)

{

fscanf(fp, "%s", r\_string);

s\_size = check\_size(r\_string);

//printf("%d %s\n", s\_size, r\_string);

size\_count = size\_count + s\_size;

if(size\_count  >= size )

my\_stop = 0;

for(int i =0; i < 10; i++)

{

if(s\_size > max\_string[i].count)

{

max\_string[i].count = s\_size;

my\_strcp(max\_string[i].string, r\_string, s\_size);

break;

}

}

}

printf("10 longest string\n");

for(int i =0; i < 10; i++)

printf("%s \n", max\_string[i].string);

fclose(fp);

return;

}
