// 6개의 숫자와 보너스 넘버 한개까지 같이 출력

#include <stdio.h>

#include <stdlib.h>

#include <time.h>

int main ()

{

int iStarter = 0;

int iGuess = 0;

int iSecret[7]={0,};

int TodySuccessCount = 0;

/- initialize random seed: \*-

srand ( time(NULL) );

do

{

if(TodySuccessCount == 10)

{

printf("Today is not good day for buy roto\n");

getchar();

return 0;

}

printf ("Guess the number (1 to 45): ");

scanf ("%d",&iGuess);

iStarter = rand() % 45 + 1;

TodySuccessCount++;

}while(iStarter != iGuess);

for(int i=0; i < 7; i++)

{

iSecret[i] = rand() % 45 +1;

for(int j=0; j<i; j++)

{

if(iSecret[i] == iSecret[j])

i--;

}

}

printf("===Roto number=== \n");

for(i =0; i< 6; i++)

{

printf("%d ",iSecret[i]);

}

printf("\n bonus : %d ",iSecret[6]);

printf("\n");

getchar();

return 0;

}
